    32→    32→export function SettingsView() {
    33→    33→  const [settings, setSettings] = useState<Settings | null>(null);
    34→    34→  const [loading, setLoading] = useState(true);
    35→    35→  const [error, setError] = useState<string | null>(null);
    36→    36→  // controlled tab: sections keep their unsaved edits while hidden
    37→    37→  const [tab, setTab] = useState("brand");
    38→    38→
    39→    39→  const { onApiError } = useAdmin();
    40→    40→  const loadStable = useCallback(async () => {
    41→    41→    setLoading(true);
    42→    42→    setError(null);
    43→    43→    try {
    44→    44→      const res = await api<{ settings: Settings }>("api/settings", { admin: true });
    45→    45→      setSettings(res.settings);
    46→    46→    } catch (err) {
    47→    47→      if (onApiError(err)) return;
    48→    48→      setError(err instanceof Error ? err.message : "Could not load settings.");
    49→    49→    } finally {
    50→    50→      setLoading(false);
    51→    51→    }
    52→    52→  }, [onApiError]);
    53→    53→
    54→    54→  useEffect(() => {
    55→    55→    void loadStable();
    56→    56→  }, [loadStable]);
    57→    57→
    58→    58→  if (loading) {
    59→    59→    return (
    60→    60→      <div>
    61→    61→        <PageHead title="Settings" desc="Everything editable on the public site." />
    62→    62→        <div className="space-y-4">
    63→    63→          <Skeleton className="h-10 w-full max-w-xl" />
    64→    64→          <Skeleton className="h-64 w-full" />
    65→    65→        </div>
    66→    66→      </div>
    67→    67→    );
    68→    68→  }
    69→    69→
    70→    70→  if (error || !settings) {
    71→    71→    return (
    72→    72→      <div className="space-y-4">
    73→    73→        <ErrorNote message={error ?? "Settings unavailable."} />
    74→    74→        <Button variant="outline" onClick={() => void loadStable()}>
    75→    75→          Try again
    76→    76→        </Button>
    77→    77→      </div>
    78→    78→    );
    79→    79→  }
    80→    80→
    81→    81→  return (
    82→    82→    <div>
    83→    83→      <PageHead title="Settings" desc="Everything editable on the public site — saved per section." />
    84→    84→      {/* forceMount keeps every section mounted so switching tabs never
    85→    85→          throws away unsaved edits in another tab */}
    86→    86→      <Tabs value={tab} onValueChange={setTab}>
    87→    87→        <TabsList className="mb-5 h-auto w-full flex-wrap justify-start gap-1 bg-ink/5 p-1">

   132→   132→/** Form state that re-seeds whenever the server value identity changes.
   133→   133→ *  Uses the documented "adjust state during render" pattern (no effect). */
   134→   134→function useSyncedForm<T>(value: T) {
   135→   135→  const [form, setForm] = useState<T>(value);
   136→   136→  const [seed, setSeed] = useState<T>(value);
   137→   137→  if (seed !== value) {
   138→   138→    setSeed(value);
   139→   139→    setForm(value);
   140→   140→  }
   141→   141→  return [form, setForm] as const;
   142→   142→}
   143→   143→
   144→   144→function useSectionSaver<K extends keyof Settings>(section: K, onSaved: (s: Settings) => void) {
   145→   145→  const { onApiError } = useAdmin();
   146→   146→  const [saving, setSaving] = useState(false);
   147→   147→
   148→   148→  async function save(data: Settings[K]) {
   149→   149→    setSaving(true);
   150→   150→    try {
   151→   151→      const res = await api<{ settings: Settings }>("api/settings", {
   152→   152→        method: "PUT",
   153→   153→        body: { section, data },
   154→   154→        admin: true,
   155→   155→      });
   156→   156→      onSaved(res.settings);
   157→   157→      toast.success("Settings saved");
   158→   158→    } catch (err) {
   159→   159→      onApiError(err);
   160→   160→    } finally {
   161→   161→      setSaving(false);
   162→   162→    }
   163→   163→  }
   164→   164→
   165→   165→  return { saving, save };
   166→   166→}

   550→   550→/* ----------------------------------------------------------------- account */
   551→   551→function AccountSection() {
   552→   552→  const [current, setCurrent] = useState("");
   553→   553→  const [next, setNext] = useState("");
   554→   554→  const [confirm, setConfirm] = useState("");
   555→   555→  const [saving, setSaving] = useState(false);
   556→   556→  const [error, setError] = useState<string | null>(null);
   557→   557→
   558→   558→  async function submit(e: FormEvent) {
   559→   559→    e.preventDefault();
   560→   560→    setError(null);
   561→   561→    if (!current) return setError("Enter your current password.");
   562→   562→    if (!PASSWORD_RE.test(next))
   563→   563→      return setError("New password needs at least 8 characters, including a letter and a number.");
   564→   564→    if (next !== confirm) return setError("New password and confirmation don't match.");
   565→   565→    setSaving(true);
   566→   566→    try {
   567→   567→      await api("api/auth", { body: { action: "password", currentPassword: current, newPassword: next }, admin: true });
   568→   568→      toast.success("Password updated");
   569→   569→      setCurrent("");
   570→   570→      setNext("");
   571→   571→      setConfirm("");
   572→   572→    } catch (err) {
   573→   573→      setError(err instanceof Error ? err.message : "Could not update password.");
   574→   574→    } finally {
   575→   575→      setSaving(false);
   576→   576→    }
   577→   577→  }
   578→   578→
   579→   579→  return (
   580→   580→    <div className="max-w-xl space-y-4">
   581→   581→      <AdminCard>
   582→   582→        <CardHead title="Change password" desc="Use at least 8 characters with a letter and a number." />
   583→   583→        <form onSubmit={(e) => void submit(e)} className="space-y-4 p-5" noValidate>
   584→   584→          <Field label="Current password" htmlFor="acc-current">
   585→   585→            <Input id="acc-current" type="password" autoComplete="current-password" value={current} onChange={(e) => setCurrent(e.target.value)} />
   586→   586→          </Field>
   587→   587→          <Field label="New password" htmlFor="acc-next">
   588→   588→            <Input id="acc-next" type="password" autoComplete="new-password" value={next} onChange={(e) => setNext(e.target.value)} />
   589→   589→          </Field>
   590→   590→          <Field label="Confirm new password" htmlFor="acc-confirm">
   591→   591→            <Input id="acc-confirm" type="password" autoComplete="new-password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
   592→   592→          </Field>
   593→   593→          {error ? (
   594→   594→            <p role="alert" className="rounded-lg border border-destructive/30 bg-destructive/5 px-3 py-2 text-sm text-destructive">
   595→   595→              {error}
   596→   596→            </p>
   597→   597→          ) : null}
   598→   598→          <Button type="submit" className="h-10 bg-ink text-ivory hover:bg-ink/85" disabled={saving}>
   599→   599→            {saving ? <Spin /> : null} Update password
   600→   600→          </Button>
   601→   601→        </form>
   602→   602→      </AdminCard>
   603→   603→    </div>
   604→   604→  );
   605→   605→}
