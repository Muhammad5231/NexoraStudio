    32→export function SettingsView() {
    33→  const [settings, setSettings] = useState<Settings | null>(null);
    34→  const [loading, setLoading] = useState(true);
    35→  const [error, setError] = useState<string | null>(null);
    36→  // controlled tab: sections keep their unsaved edits while hidden
    37→  const [tab, setTab] = useState("brand");
    38→
    39→  const { onApiError } = useAdmin();
    40→  const loadStable = useCallback(async () => {
    41→    setLoading(true);
    42→    setError(null);
    43→    try {
    44→      const res = await api<{ settings: Settings }>("api/settings", { admin: true });
    45→      setSettings(res.settings);
    46→    } catch (err) {
    47→      if (onApiError(err)) return;
    48→      setError(err instanceof Error ? err.message : "Could not load settings.");
    49→    } finally {
    50→      setLoading(false);
    51→    }
    52→  }, [onApiError]);
    53→
    54→  useEffect(() => {
    55→    void loadStable();
    56→  }, [loadStable]);
    57→
    58→  if (loading) {
    59→    return (
    60→      <div>
    61→        <PageHead title="Settings" desc="Everything editable on the public site." />
    62→        <div className="space-y-4">
    63→          <Skeleton className="h-10 w-full max-w-xl" />
    64→          <Skeleton className="h-64 w-full" />
    65→        </div>
    66→      </div>
    67→    );
    68→  }
    69→
    70→  if (error || !settings) {
    71→    return (
    72→      <div className="space-y-4">
    73→        <ErrorNote message={error ?? "Settings unavailable."} />
    74→        <Button variant="outline" onClick={() => void loadStable()}>
    75→          Try again
    76→        </Button>
    77→      </div>
    78→    );
    79→  }
    80→
    81→  return (
    82→    <div>
    83→      <PageHead title="Settings" desc="Everything editable on the public site — saved per section." />
    84→      {/* forceMount keeps every section mounted so switching tabs never
    85→          throws away unsaved edits in another tab */}
    86→      <Tabs value={tab} onValueChange={setTab}>
    87→        <TabsList className="mb-5 h-auto w-full flex-wrap justify-start gap-1 bg-ink/5 p-1">
    88→          <TabsTrigger value="brand">Brand</TabsTrigger>
    89→          <TabsTrigger value="contact">Contact</TabsTrigger>
    90→          <TabsTrigger value="social">Social</TabsTrigger>
    91→          <TabsTrigger value="seo">SEO</TabsTrigger>
    92→          <TabsTrigger value="home">Home</TabsTrigger>
    93→          <TabsTrigger value="about">About</TabsTrigger>
    94→          <TabsTrigger value="footer">Footer</TabsTrigger>
    95→          <TabsTrigger value="pages">Pages</TabsTrigger>
    96→          <TabsTrigger value="account">Account</TabsTrigger>
    97→        </TabsList>
    98→
    99→        <TabsContent value="brand" forceMount className={tab === "brand" ? undefined : "hidden"}>
   100→          <BrandSection value={settings.brand} onSaved={setSettings} />
   101→        </TabsContent>
   102→        <TabsContent value="contact" forceMount className={tab === "contact" ? undefined : "hidden"}>
   103→          <ContactSection value={settings.contact} onSaved={setSettings} />
   104→        </TabsContent>
   105→        <TabsContent value="social" forceMount className={tab === "social" ? undefined : "hidden"}>
   106→          <SocialSection value={settings.social} onSaved={setSettings} />
   107→        </TabsContent>
   108→        <TabsContent value="seo" forceMount className={tab === "seo" ? undefined : "hidden"}>
   109→          <SeoSection value={settings.seo} onSaved={setSettings} />
   110→        </TabsContent>
   111→        <TabsContent value="home" forceMount className={tab === "home" ? undefined : "hidden"}>
   112→          <HomeSection value={settings.home} onSaved={setSettings} />
   113→        </TabsContent>
   114→        <TabsContent value="about" forceMount className={tab === "about" ? undefined : "hidden"}>
   115→          <AboutSection value={settings.about} onSaved={setSettings} />
   116→        </TabsContent>

   131→/* ------------------------------------------------------------------ shared */
   132→/** Form state that re-seeds whenever the server value identity changes.
   133→ *  Uses the documented "adjust state during render" pattern (no effect). */
   134→function useSyncedForm<T>(value: T) {
   135→  const [form, setForm] = useState<T>(value);
   136→  const [seed, setSeed] = useState<T>(value);
   137→  if (seed !== value) {
   138→    setSeed(value);
   139→    setForm(value);
   140→  }
   141→  return [form, setForm] as const;
   142→}
   143→
   144→function useSectionSaver<K extends keyof Settings>(section: K, onSaved: (s: Settings) => void) {
   145→  const { onApiError } = useAdmin();
   146→  const [saving, setSaving] = useState(false);
   147→
   148→  async function save(data: Settings[K]) {
   149→    setSaving(true);
   150→    try {
   151→      const res = await api<{ settings: Settings }>("api/settings", {
   152→        method: "PUT",
   153→        body: { section, data },
   154→        admin: true,
   155→      });
   156→      onSaved(res.settings);
   157→      toast.success("Settings saved");
   158→    } catch (err) {
   159→      onApiError(err);
   160→    } finally {
   161→      setSaving(false);
   162→    }
   163→  }
   164→
   165→  return { saving, save };
   166→}

   550→/* ----------------------------------------------------------------- account */
   551→function AccountSection() {
   552→  const [current, setCurrent] = useState("");
   553→  const [next, setNext] = useState("");
   554→  const [confirm, setConfirm] = useState("");
   555→  const [saving, setSaving] = useState(false);
   556→  const [error, setError] = useState<string | null>(null);
   557→
   558→  async function submit(e: FormEvent) {
   559→    e.preventDefault();
   560→    setError(null);
   561→    if (!current) return setError("Enter your current password.");
   562→    if (!PASSWORD_RE.test(next))
   563→      return setError("New password needs at least 8 characters, including a letter and a number.");
   564→    if (next !== confirm) return setError("New password and confirmation don't match.");
   565→    setSaving(true);
   566→    try {
   567→      await api("api/auth", { body: { action: "password", currentPassword: current, newPassword: next }, admin: true });
   568→      toast.success("Password updated");
   569→      setCurrent("");
   570→      setNext("");
   571→      setConfirm("");
   572→    } catch (err) {
   573→      setError(err instanceof Error ? err.message : "Could not update password.");
   574→    } finally {
   575→      setSaving(false);
   576→    }
   577→  }
   578→
   579→  return (
   580→    <div className="max-w-xl space-y-4">
   581→      <AdminCard>
   582→        <CardHead title="Change password" desc="Use at least 8 characters with a letter and a number." />
   583→        <form onSubmit={(e) => void submit(e)} className="space-y-4 p-5" noValidate>
   584→          <Field label="Current password" htmlFor="acc-current">
   585→            <Input id="acc-current" type="password" autoComplete="current-password" value={current} onChange={(e) => setCurrent(e.target.value)} />
   586→          </Field>
   587→          <Field label="New password" htmlFor="acc-next">
   588→            <Input id="acc-next" type="password" autoComplete="new-password" value={next} onChange={(e) => setNext(e.target.value)} />
   589→          </Field>
   590→          <Field label="Confirm new password" htmlFor="acc-confirm">
   591→            <Input id="acc-confirm" type="password" autoComplete="new-password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
   592→          </Field>
   593→          {error ? (
   594→            <p role="alert" className="rounded-lg border border-destructive/30 bg-destructive/5 px-3 py-2 text-sm text-destructive">
   595→              {error}
   596→            </p>
   597→          ) : null}
   598→          <Button type="submit" className="h-10 bg-ink text-ivory hover:bg-ink/85" disabled={saving}>
   599→            {saving ? <Spin /> : null} Update password
   600→          </Button>
   601→        </form>
   602→      </AdminCard>
   603→    </div>
   604→  );
   605→}
