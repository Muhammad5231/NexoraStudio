    30→/* ------------------------------------------------------------------ Reveal */

    32→export type RevealVariant = "up" | "blur" | "scale" | "left" | "right";
    33→
    34→/**
    35→ * Scroll-triggered entrance, driven by ONE shared IntersectionObserver + CSS
    36→ * transitions (see src/lib/in-view.ts + the [data-reveal] rules in
    37→ * globals.css). Replaces the old framer-motion whileInView version whose
    38→ * per-element observers could miss and leave content stuck at opacity:0.
    39→ *
    40→ * Failure-proof by design — the hidden state is applied by JS only and has
    41→ * four safety nets, so the worst case is "appears without animation",
    42→ * never "invisible forever":
    43→ *   1. SSR markup is fully visible (no opacity:0 in the HTML — no-JS safe).
    44→ *   2. Without IntersectionObserver support nothing is ever hidden.
    45→ *   3. prefers-reduced-motion forces visibility in CSS regardless of JS.
    46→ *   4. A watchdog reveals any element still hidden while on screen.
    47→ */
    48→export function Reveal({
    49→  children,
    50→  delay = 0,
    51→  y = 28,
    52→  className,
    53→  once = true,
    54→  variant = "up",
    55→}: {
    56→  children: ReactNode;
    57→  delay?: number;
    58→  y?: number;
    59→  className?: string;
    60→  /** Kept for API compatibility — reveals are permanent (no consumers re-hide). */
    61→  once?: boolean;
    62→  variant?: RevealVariant;
    63→}) {
    64→  const ref = useRef<HTMLDivElement>(null);
    65→
    66→  useIsoLayoutEffect(() => {
    67→    const el = ref.current;
    68→    if (!el) return;
    69→    // reduced motion / old browsers: keep content fully visible (CSS also
    70→    // enforces this — see the prefers-reduced-motion block in globals.css)
    71→    if (typeof IntersectionObserver === "undefined") return;
    72→
    73→    // hidden state lives in dataset attributes, which React never rewrites
    74→    // (unlike classList — React re-applies className on prop changes)
    75→    el.dataset.reveal = variant;
    76→    if (variant === "up" && y !== 28) el.style.setProperty("--reveal-y", `${y}px`);
    77→    if (delay > 0) el.style.setProperty("--reveal-delay", `${Math.round(delay * 1000)}ms`);
    78→    el.getBoundingClientRect(); // commit the hidden state so the transition runs even if already on screen
    79→
    80→    let revealed = false;
    81→    let settleTimer = 0;
    82→    const show = () => {
    83→      if (revealed) return;
    84→      revealed = true;
    85→      el.dataset.revealState = "in";
    86→      if (once) stop();
    87→      // once the transition settles, strip the hooks so the element is back
    88→      // to plain CSS (no lingering transition-delay on hover styles etc.)
    89→      window.clearTimeout(settleTimer);
    90→      settleTimer = window.setTimeout(() => {
    91→        delete el.dataset.revealState;
    92→        delete el.dataset.reveal;
    93→        el.style.removeProperty("--reveal-delay");
    94→        el.style.removeProperty("--reveal-y");
    95→      }, 900 + delay * 1000);
    96→    };
    97→    const stop = onEnterView(el, show);
    98→    // safety net: if the event never fired but the element is on screen, show it
    99→    const watchdog = window.setTimeout(() => {
   100→      if (!revealed && isInViewport(el)) show();
   101→    }, 1200);
   102→
   103→    return () => {
   104→      window.clearTimeout(watchdog);
   105→      window.clearTimeout(settleTimer);
   106→      stop();
   107→      delete el.dataset.revealState;
   108→      delete el.dataset.reveal;
   109→      el.style.removeProperty("--reveal-delay");
   110→      el.style.removeProperty("--reveal-y");
   111→    };
   112→  }, [variant, delay, y, once]);
   113→
   114→  return (
   115→    <div ref={ref} className={className}>
   116→      {children}
   117→    </div>
   118→  );
   119→}

   121→/* ------------------------------------------------------------- WordsReveal */
   122→/**
   123→ * Word-by-word masked reveal driven by scroll position — each word slides up
   124→ * out of its own overflow mask as the text enters the viewport.
   125→ *
   126→ * Same failure-proof pattern as Reveal: SSR markup renders the words fully
   127→ * visible; the hidden state is added by JS only (data-words-pre, CSS-driven
   128→ * stagger via --wr-i), the shared observer flips it, and a watchdog plus the
   129→ * prefers-reduced-motion CSS rule guarantee the text can never stay hidden.
   130→ */
   131→export function WordsReveal({
   132→  text,
   133→  className,
   134→  wordClassName,
   135→  delay = 0,
   136→  stagger = 0.05,
   137→  once = true,
   138→}: {
   139→  text: string;
   140→  className?: string;
   141→  wordClassName?: string;
   142→  delay?: number;
   143→  stagger?: number;
   144→  /** Kept for API compatibility — reveals are permanent (no consumers re-hide). */
   145→  once?: boolean;
   146→}) {
   147→  const ref = useRef<HTMLSpanElement>(null);
   148→  const words = useMemo(() => text.split(/\s+/).filter(Boolean), [text]);
   149→
   150→  useIsoLayoutEffect(() => {
   151→    const el = ref.current;
   152→    if (!el) return;
   153→    if (typeof IntersectionObserver === "undefined") return;
   154→    // observe the un-clipped container — masked inner spans are fully clipped
   155→    // by their overflow-hidden parents, so observing them directly can miss the
   156→    // "already in view at mount" case (e.g. hash-navigating to a new view)
   157→    el.dataset.wordsPre = "true";
   158→    el.style.setProperty("--wr-delay", `${Math.round(delay * 1000)}ms`);
   159→    el.style.setProperty("--wr-stagger", `${Math.round(stagger * 1000)}ms`);
   160→    el.getBoundingClientRect(); // commit the hidden state before revealing
   161→
   162→    let revealed = false;
   163→    let settleTimer = 0;
   164→    const show = () => {
   165→      if (revealed) return;
   166→      revealed = true;
   167→      delete el.dataset.wordsPre;
   168→      if (once) stop();
   169→      window.clearTimeout(settleTimer);
   170→      settleTimer = window.setTimeout(() => {
   171→        el.style.removeProperty("--wr-delay");
   172→        el.style.removeProperty("--wr-stagger");
   173→      }, 900 + (delay + words.length * stagger) * 1000);
   174→    };
   175→    const stop = onEnterView(el, show);
   176→    const watchdog = window.setTimeout(() => {
   177→      if (!revealed && isInViewport(el)) show();
   178→    }, 1200);
   179→
   180→    return () => {
   181→      window.clearTimeout(watchdog);
   182→      window.clearTimeout(settleTimer);
   183→      stop();
   184→      delete el.dataset.wordsPre;
   185→      el.style.removeProperty("--wr-delay");
   186→      el.style.removeProperty("--wr-stagger");
   187→    };
   188→  }, [text, delay, stagger, once, words.length]);
   189→
   190→  return (
   191→    <span ref={ref} className={cn("words-reveal", className)}>
   192→      {words.map((word, i) => (
   193→        <span
   194→          key={`${word}-${i}`}
   195→          className="inline-block overflow-hidden pb-[0.12em] -mb-[0.12em] align-bottom"
   196→        >
   197→          <span
   198→            className={cn("words-reveal-word inline-block", wordClassName)}
   199→            style={{ "--wr-i": i } as CSSProperties}
   200→          >
   201→            {word}
   202→            {i < words.length - 1 ? "\u00A0" : ""}
   203→          </span>
   204→        </span>
   205→      ))}
   206→    </span>
   207→  );
   208→}

   210→/* ---------------------------------------------------------- ScrollTextFill */
   211→/**
   212→ * Manifesto-style fill: words start dim and ink themselves in one by one as
   213→ * the paragraph travels up the viewport (scroll-linked, not time-based).
   214→ */
   215→export function ScrollTextFill({ text, className }: { text: string; className?: string }) {
   216→  const ref = useRef<HTMLParagraphElement>(null);
   217→  const reduce = useReducedMotion();
   218→  const { scrollYProgress } = useScroll({ target: ref, offset: ["start 0.92", "end 0.55"] });
   219→  const words = useMemo(() => text.split(/\s+/).filter(Boolean), [text]);
   220→
   221→  return (
   222→    <p ref={ref} className={cn("relative", className)}>
   223→      {words.map((word, i) => (
   224→        <FillWord
   225→          key={`${word}-${i}`}
   226→          progress={scrollYProgress}
   227→          range={[i / words.length, Math.min(1, (i + 1.6) / words.length)]}
   228→          dim={reduce === true}
   229→        >
   230→          {word}
   231→        </FillWord>
   232→      ))}
   233→    </p>
   234→  );
   235→}
   236→
   237→function FillWord({
   238→  progress,
   239→  range,
   240→  dim,
   241→  children,
   242→}: {
   243→  progress: MotionValue<number>;
   244→  range: [number, number];
   245→  dim: boolean;
   246→  children: string;
   247→}) {
   248→  const opacity = useTransform(progress, range, dim ? [1, 1] : [0.14, 1]);
   249→  return (
   250→    <motion.span style={{ opacity }} className="mr-[0.26em] inline-block">
   251→      {children}
   252→    </motion.span>
   253→  );
   254→}

   467→/* ---------------------------------------------------------------- Magnetic */
   468→export function Magnetic({ children, strength = 0.25, className }: { children: ReactNode; strength?: number; className?: string }) {
   469→  const ref = useRef<HTMLDivElement>(null);
   470→  const x = useMotionValue(0);
   471→  const y = useMotionValue(0);
   472→  const sx = useSpring(x, { stiffness: 180, damping: 14, mass: 0.4 });
   473→  const sy = useSpring(y, { stiffness: 180, damping: 14, mass: 0.4 });
   474→
   475→  return (
   476→    <motion.div
   477→      ref={ref}
   478→      className={cn("inline-block", className)}
   479→      style={{ x: sx, y: sy }}
   480→      onMouseMove={(e) => {
   481→        if (window.matchMedia("(pointer: coarse)").matches) return;
   482→        const rect = ref.current?.getBoundingClientRect();
   483→        if (!rect) return;
   484→        x.set((e.clientX - rect.left - rect.width / 2) * strength);
   485→        y.set((e.clientY - rect.top - rect.height / 2) * strength);
   486→      }}
   487→      onMouseLeave={() => {
   488→        x.set(0);
   489→        y.set(0);
   490→      }}
   491→    >
   492→      {children}
   493→    </motion.div>
   494→  );
   495→}

   518→/* ----------------------------------------------------------------- Counter */
   519→export function Counter({ value, suffix, className }: { value: string; suffix?: string; className?: string }) {
   520→  const ref = useRef<HTMLSpanElement>(null);
   521→  const reduce = useReducedMotion();
   522→  const numeric = parseFloat(value);
   523→  const isNumeric = !isNaN(numeric) && isFinite(numeric);
   524→  const [display, setDisplay] = useState(isNumeric ? "0" : value);
   525→  const [seed, setSeed] = useState(value);
   526→  const [inView, setInView] = useState(false);
   527→
   528→  // resync when the value prop changes (e.g. content refetch after admin
   529→  // edits) — "adjust state during render" pattern, no effect needed
   530→  if (seed !== value) {
   531→    setSeed(value);
   532→    setDisplay(isNumeric ? "0" : value);
   533→  }
   534→
   535→  // shared observer (one for the whole app) instead of a per-element
   536→  // framer useInView — callback is always async, so no sync setState
   537→  useEffect(() => {
   538→    const el = ref.current;
   539→    if (!el) return;
   540→    const stop = onEnterView(el, () => {
   541→      stop();
   542→      setInView(true);
   543→    });
   544→    return stop;
   545→  }, []);
   546→
   547→  useEffect(() => {
   548→    if (!inView || !isNumeric) return;
   549→    let raf = 0;
   550→    // reduced motion: show the final value instantly (next frame — avoids
   551→    // synchronous setState in the effect; render stays hydration-stable,
   552→    // server and client both render "0" first)
   553→    if (reduce) {
   554→      raf = requestAnimationFrame(() => setDisplay(String(numeric)));
   555→      return () => cancelAnimationFrame(raf);
   556→    }
   557→    const duration = 1200;
   558→    const start = performance.now();
   559→    const tick = (now: number) => {
   560→      const p = Math.min(1, (now - start) / duration);
   561→      const eased = 1 - Math.pow(1 - p, 3);
   562→      setDisplay(String(Math.round(numeric * eased)));
   563→      if (p < 1) raf = requestAnimationFrame(tick);
   564→    };
   565→    raf = requestAnimationFrame(tick);
   566→    return () => cancelAnimationFrame(raf);
   567→  }, [inView, isNumeric, numeric, reduce]);
   568→
   569→  return (
   570→    <span ref={ref} className={className}>
   571→      {display}
   572→      {suffix}
   573→    </span>
   574→  );
   575→}

   577→/* -------------------------------------------------------------- ParallaxImg */
   578→export function ParallaxImg({
   579→  src,
   580→  alt,
   581→  className,
   582→  imgClassName,
   583→  distance = 40,
   584→  priority = false,
   585→}: {
   586→  src: string;
   587→  alt: string;
   588→  className?: string;
   589→  imgClassName?: string;
   590→  distance?: number;
   591→  priority?: boolean;
   592→}) {
   593→  const ref = useRef<HTMLDivElement>(null);
   594→  const reduce = useReducedMotion();
   595→  const { scrollYProgress } = useScroll({ target: ref, offset: ["start end", "end start"] });
   596→  const y = useTransform(scrollYProgress, [0, 1], [-distance, distance]);
   597→
   598→  return (
   599→    <div ref={ref} className={cn("relative overflow-hidden", className)}>
   600→      <motion.img
   601→        src={src}
   602→        alt={alt}
   603→        style={reduce ? undefined : { y, scale: 1.18 }}
   604→        className={cn("h-full w-full object-cover", imgClassName)}
   605→        loading={priority ? "eager" : "lazy"}
   606→      />
   607→    </div>
   608→  );
   609→}
