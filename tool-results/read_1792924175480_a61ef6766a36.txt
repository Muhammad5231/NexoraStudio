    68→function missingListApiNote(status: number): string | null {
    69→  if (status === 404 || status === 405) {
    70→    return "The enquiries list endpoint (GET /api/enquiries) isn't available yet — the backend only exposes the public POST + per-lead detail routes. Lead detail, status, notes and delete still work from a direct link.";
    71→  }
    72→  return null;

    96→  useEffect(() => {

   118→  async function patch(body: Record<string, unknown>, optimist?: (l: Enquiry) => Enquiry) {
   119→    if (!lead) return;

   122→      const next = optimist(prev);

   124→      onUpdated(next);

   127→      const res = await api<{ enquiry: Enquiry }>(`api/enquiries/${lead.id}`, {
   128→        method: "PATCH",
   129→        body,
   130→        admin: true,
   131→      });
   132→      setLead(res.enquiry);
   133→      onUpdated(res.enquiry);
   134→    } catch (err) {
   135→      setLead(prev); // revert optimistic update (detail + list)
   136→      onUpdated(prev);
   137→      onApiError(err);
   138→    }

   196→            <Skeleton className="h-32" />
   197→            <Skeleton className="h-40" />
   198→          </div>
   199→        ) : (
   200→          <div className="space-y-6 p-5">
   201→            {/* contact block */}
   202→            <section aria-label="Contact details" className="space-y-2">

   204→                <a
   205→                  href={`mailto:${lead.email}`}
   206→                  className="inline-flex h-9 items-center gap-2 rounded-lg border border-ink/15 bg-white px-3 text-sm font-medium text-ink transition-colors hover:bg-ink/[0.04]"
   207→                >
   208→                  <Mail className="size-4" aria-hidden="true" /> Email
   209→                </a>

   211→                  <a

   216→                  >
   217→                    <MessageCircle className="size-4" aria-hidden="true" /> WhatsApp
   218→                  </a>

   371→              </div>
   372→            </section>

   380→                variant="outline"
   381→                size="sm"
   382→                className="h-9 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   383→                onClick={() => setConfirmDelete(true)}

   385→                <Trash2 className="size-4" /> Delete lead

   387→            </section>
   388→          </div>

   390→
   391→        <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>

   393→            <AlertDialogHeader>

   395→              <AlertDialogDescription>

   397→              </AlertDialogDescription>

   399→            <AlertDialogFooter>
   400→              <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   401→              <AlertDialogAction
   402→                disabled={deleting}
   403→                className="bg-destructive text-white hover:bg-destructive/90"

   406→                  void deleteLead();

   409→                {deleting ? <Spin /> : null} Delete

   411→            </AlertDialogFooter>
   412→          </AlertDialogContent>
   413→        </AlertDialog>
   414→      </SheetContent>
   415→    </Sheet>
   416→  );
   417→}

   420→export function LeadsView({ selectedId }: { selectedId?: string }) {
   421→  const { onApiError } = useAdmin();
   422→
   423→  const [items, setItems] = useState<Enquiry[]>([]);
   424→  const [total, setTotal] = useState(0);
   425→  const [page, setPage] = useState(1);
   426→  const [loading, setLoading] = useState(true);
   427→  const [error, setError] = useState<string | null>(null);
   428→  const [hint, setHint] = useState<string | null>(null);
   429→
   430→  const [search, setSearch] = useState("");
   431→  const [status, setStatus] = useState("all");
   432→  const [service, setService] = useState("all");
   433→  const [sort, setSort] = useState<"newest" | "oldest">("newest");
   434→  const debouncedSearch = useDebouncedValue(search, 300);
   435→
   436→  const [services, setServices] = useState<Service[]>([]);
   437→
   438→  useEffect(() => {
   439→    (async () => {
   440→      try {
   441→        const res = await api<{ services: Service[] }>("api/services?all=1", { admin: true });
   442→        setServices(res.services);
   443→      } catch {
   444→        /* filter options are optional — ignore */
   445→      }
   446→    })();
   447→  }, []);
   448→
   449→  // reset to first page whenever filters change
   450→  useEffect(() => {
   451→    setPage(1);
   452→  }, [debouncedSearch, status, service, sort]);
   453→
   454→  const queryString = useMemo(() => {
   455→    const params = new URLSearchParams();
   456→    params.set("page", String(page));
   457→    params.set("pageSize", String(PAGE_SIZE));
   458→    if (debouncedSearch.trim()) params.set("search", debouncedSearch.trim());
   459→    if (status !== "all") params.set("status", status);
   460→    if (service !== "all") params.set("service", service);
   461→    params.set("sort", sort);
   462→    return params.toString();
   463→  }, [page, debouncedSearch, status, service, sort]);
   464→
   465→  const load = useCallback(async () => {
   466→    setLoading(true);
   467→    setError(null);
   468→    setHint(null);
   469→    try {
   470→      const res = await api<Paginated<Enquiry>>(`api/enquiries?${queryString}`, { admin: true });
   471→      setItems(res.items);
   472→      setTotal(res.total);
   473→    } catch (err) {
   474→      if (onApiError(err)) return;
   475→      const status = err instanceof Error && "status" in err ? Number((err as { status: number }).status) : 0;
   476→      setError(err instanceof Error ? err.message : "Could not load leads.");
   477→      const miss = typeof status === "number" ? missingListApiNote(status) : null;
   478→      if (miss) setHint(miss);
   479→    } finally {
   480→      setLoading(false);
   481→    }
   482→  }, [queryString, onApiError]);
   483→
   484→  useEffect(() => {
   485→    void load();
   486→  }, [load]);
   487→
   488→  function onUpdated(updated: Enquiry) {
   489→    setItems((prev) => patchInList(prev, updated));
   490→  }
   491→
   492→  function onDeleted(id: string) {
   493→    setItems((prev) => prev.filter((e) => e.id !== id));
   494→    setTotal((t) => Math.max(0, t - 1));
   495→    void load();
   496→  }
   497→
   498→  const from = total === 0 ? 0 : (page - 1) * PAGE_SIZE + 1;
   499→  const to = Math.min(page * PAGE_SIZE, total);
   500→  const maxPage = Math.max(1, Math.ceil(total / PAGE_SIZE));
   501→
   502→  const filterControls = (
   503→    <div className="flex flex-col gap-2 sm:flex-row sm:flex-wrap sm:items-center">
   504→      <div className="relative">
   505→        <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-ink/35" aria-hidden="true" />
   506→        <Input
   507→          value={search}
   508→          onChange={(e) => setSearch(e.target.value)}
   509→          placeholder="Search name, email, business…"
   510→          aria-label="Search leads"
   511→          className="h-10 w-full pl-9 sm:w-64"
   512→        />
   513→      </div>
   514→      <Select value={status} onValueChange={setStatus}>
   515→        <SelectTrigger className="h-10 w-full sm:w-40" aria-label="Filter by status">
   516→          <SelectValue />
   517→        </SelectTrigger>
   518→        <SelectContent>
   519→          <SelectItem value="all">All statuses</SelectItem>
   520→          {ENQUIRY_STATUSES.map((s) => (
   521→            <SelectItem key={s} value={s}>
   522→              {ENQUIRY_STATUS_LABELS[s]}
   523→            </SelectItem>
   524→          ))}
   525→        </SelectContent>
   526→      </Select>
   527→      <Select value={service} onValueChange={setService}>
   528→        <SelectTrigger className="h-10 w-full sm:w-44" aria-label="Filter by service">
   529→          <SelectValue />
   530→        </SelectTrigger>
   531→        <SelectContent>
   532→          <SelectItem value="all">All services</SelectItem>
   533→          {services.map((s) => (
   534→            <SelectItem key={s.id} value={s.title}>
   535→              {s.title}
   536→            </SelectItem>
   537→          ))}
   538→        </SelectContent>
   539→      </Select>
   540→      <Select value={sort} onValueChange={(v) => setSort(v === "oldest" ? "oldest" : "newest")}>
   541→        <SelectTrigger className="h-10 w-full sm:w-36" aria-label="Sort order">
   542→          <SelectValue />
   543→        </SelectTrigger>
   544→        <SelectContent>
   545→          <SelectItem value="newest">Newest first</SelectItem>
   546→          <SelectItem value="oldest">Oldest first</SelectItem>
   547→        </SelectContent>
   548→      </Select>
   549→    </div>
   550→  );
   551→
   552→  return (
   553→    <div>
   554→      <PageHead title="Leads" desc="Every enquiry from your public contact form, in one inbox." />
   555→
   556→      {hint ? (
   557→        <div className="mb-4">
   558→          <ErrorNote message={hint} />
   559→        </div>
   560→      ) : null}
   561→
   562→      <AdminCard>
   563→        <div className="border-b border-ink/10 p-4">{filterControls}</div>
   564→
   565→        {loading && items.length === 0 ? (
   566→          <div className="p-4">
   567→            <ListSkeleton rows={5} />
   568→          </div>
   569→        ) : error && items.length === 0 ? (
   570→          <div className="space-y-4 p-4">
   571→            <ErrorNote message={error} />
   572→            <Button variant="outline" onClick={() => void load()}>
   573→              Try again
   574→            </Button>
   575→          </div>
   576→        ) : items.length === 0 ? (
   577→          <EmptyState
   578→            icon={<Inbox className="size-6" aria-hidden="true" />}
   579→            title={search || status !== "all" || service !== "all" ? "No leads match your filters" : "No enquiries yet"}
   580→            body={
   581→              search || status !== "all" || service !== "all"
   582→                ? "Try a different search or clear the filters."
   583→                : "They'll appear here the moment your contact form receives one."
   584→            }
   585→            action={
   586→              search || status !== "all" || service !== "all" ? (
   587→                <Button
   588→                  variant="outline"
   589→                  onClick={() => {
   590→                    setSearch("");
   591→                    setStatus("all");
   592→                    setService("all");
   593→                  }}
   594→                >
   595→                  Clear filters
   596→                </Button>
   597→              ) : undefined
   598→            }
   599→          />
   600→        ) : (
   601→          <>
   602→            {/* desktop table */}
   603→            <div className={cn("hidden overflow-x-auto md:block", loading && "opacity-60")} aria-busy={loading}>
   604→              <Table>
   605→                <TableHeader>
