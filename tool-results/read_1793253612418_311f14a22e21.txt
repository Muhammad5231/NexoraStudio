   103→/** Cursor-following volt glow — fine pointers only, purely decorative. */
   104→function Spotlight() {
   105→  const reduce = useReducedMotion();
   106→  const wrapRef = useRef<HTMLDivElement>(null);
   107→  const primed = useRef(false);
   108→  const x = useMotionValue(-900);
   109→  const y = useMotionValue(-900);
   110→  const sx = useSpring(x, { stiffness: 55, damping: 20, mass: 0.6 });
   111→  const sy = useSpring(y, { stiffness: 55, damping: 20, mass: 0.6 });
   112→
   113→  useEffect(() => {
   114→    if (reduce || typeof window === "undefined") return;
   115→    if (!window.matchMedia("(pointer: fine)").matches) return;
   116→    const onMove = (e: MouseEvent) => {
   117→      const rect = wrapRef.current?.getBoundingClientRect();
   118→      if (!rect) return;
   119→      const px = e.clientX - rect.left;
   120→      const py = e.clientY - rect.top;
   121→      if (!primed.current) {
   122→        // first move: snap to the cursor so the glow doesn't fly across the hero
   123→        primed.current = true;
   124→        sx.jump(px);
   125→        sy.jump(py);
   126→        return;
   127→      }
   128→      x.set(px);
   129→      y.set(py);
   130→    };
   131→    window.addEventListener("mousemove", onMove, { passive: true });
   132→    return () => window.removeEventListener("mousemove", onMove);
   133→  }, [reduce, x, y, sx, sy]);
   134→
   135→  return (
   136→    <div ref={wrapRef} aria-hidden="true" className="pointer-events-none absolute inset-0 z-[2] overflow-hidden">
   137→      <motion.div style={{ x: sx, y: sy }} className="absolute left-0 top-0 will-change-transform">
   138→        <div className="h-[560px] w-[560px] -translate-x-1/2 -translate-y-1/2 rounded-full bg-[radial-gradient(circle,rgba(198,255,58,0.07)_0%,rgba(198,255,58,0.02)_38%,transparent_68%)]" />
   139→      </motion.div>
   140→    </div>
   141→  );
   142→}

   719→function ProcessSection() {
   720→  const { settings } = useSite();
   721→  const h = settings.home;
   722→  const lineRef = useRef<HTMLDivElement>(null);
   723→  const reduce = useReducedMotion();
   724→  const { scrollYProgress } = useScroll({ target: lineRef, offset: ["start 0.85", "end 0.55"] });
   725→  const lineScale = useSpring(scrollYProgress, { stiffness: 90, damping: 22 });
   726→
   727→  return (
   728→    <section className="bg-ink py-20 sm:py-28">
   729→      <div className="container-x">
   730→        <SectionHead kicker={h.processKicker} title={h.processTitle} align="center" />
   731→
   732→        <div ref={lineRef} className="relative mt-16 grid gap-10 sm:grid-cols-2 lg:grid-cols-4 lg:gap-8">
   733→          {/* connecting line (desktop) — base track */}
   734→          <div className="absolute left-0 right-0 top-7 hidden h-px bg-ivory/10 lg:block" aria-hidden="true" />
   735→          {/* volt progress line — draws itself across the steps as you scroll */}
   736→          {reduce ? null : (
   737→            <motion.div
   738→              style={{ scaleX: lineScale }}
   739→              className="absolute left-0 right-0 top-7 hidden h-px origin-left bg-gradient-to-r from-volt via-volt/70 to-volt/15 lg:block"
   740→              aria-hidden="true"
   741→            />
   742→          )}
   743→          {h.processSteps.map((step, i) => (
   744→            <Reveal key={i} delay={0.08 * i} variant="blur">
