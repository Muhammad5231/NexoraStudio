   127→  const [projects, setProjects] = useState<Project[]>([]);

   131→  const [deleting, setDeleting] = useState(false);
   132→
   133→  const load = useCallback(async () => {
   134→    setLoading(true);
   135→    setError(null);
   136→    try {
   137→      const res = await api<{ projects: Project[] }>("api/projects?all=1", { admin: true });
   138→      setProjects(res.projects);
   139→    } catch (err) {
   140→      if (onApiError(err)) return;
   141→      setError(err instanceof Error ? err.message : "Could not load projects.");

   143→      setLoading(false);

   147→  useEffect(() => {

   202→                <TableHeader>
   203→                  <TableRow className="hover:bg-transparent">

   343→/* =================================================================== Editor */
   344→export function ProjectEditor({ id }: { id?: string }) {
   345→  const { onApiError } = useAdmin();
   346→  const isNew = !id;
   347→
   348→  const [form, setForm] = useState<ProjectForm>(emptyForm);
   349→  const [slugTouched, setSlugTouched] = useState(false);
   350→  const [loading, setLoading] = useState(!isNew);
   351→  const [loadError, setLoadError] = useState<string | null>(null);
   352→  const [saving, setSaving] = useState(false);
   353→  const [deleting, setDeleting] = useState(false);
   354→  const [confirmDelete, setConfirmDelete] = useState(false);
   355→
   356→  const set = <K extends keyof ProjectForm>(key: K, value: ProjectForm[K]) =>
   357→    setForm((f) => ({ ...f, [key]: value }));
   358→
   359→  useEffect(() => {
   360→    if (!id) return;
   361→    let alive = true;
   362→    setLoading(true);
   363→    (async () => {
   364→      try {
   365→        const res = await api<{ project: Project }>(`api/projects/${id}`, { admin: true });
   366→        if (!alive) return;
   367→        setForm(toForm(res.project));
   368→        setSlugTouched(true); // keep existing slug as-is
   369→      } catch (err) {
   370→        if (onApiError(err)) return;
   371→        setLoadError(err instanceof Error ? err.message : "Could not load this project.");
   372→      } finally {
   373→        if (alive) setLoading(false);
   374→      }
   375→    })();
   376→    return () => {
   377→      alive = false;
   378→    };
   379→  }, [id, onApiError]);
   380→
   381→  const slugError = useMemo(() => {
   382→    if (!form.slug) return "Slug is required — it becomes the public URL.";
   383→    if (!SLUG_RE.test(form.slug)) return "Use lowercase letters, numbers and dashes only.";
   384→    return null;
   385→  }, [form.slug]);
   386→
   387→  async function save(opts?: { publish?: boolean }) {
   388→    if (form.title.trim().length < 2) {
   389→      toast.error("Give the project a title (at least 2 characters).");
   390→      return;
   391→    }
   392→    if (slugError) {
   393→      toast.error(slugError);
   394→      return;
   395→    }
   396→    const effective: ProjectForm = { ...form, published: opts?.publish ? true : form.published };
   397→    const body = {
   398→      ...effective,
   399→      title: effective.title.trim(),
   400→      slug: effective.slug.trim(),
   401→      client: effective.client.trim(),
   402→      year: effective.year.trim(),
   403→      liveUrl: effective.liveUrl.trim(),
   404→      coverImage: effective.coverImage.trim(),
   405→      objectives: effective.objectives.map((s) => s.trim()).filter(Boolean),
   406→      features: effective.features.map((s) => s.trim()).filter(Boolean),
   407→      technologies: effective.technologies.map((s) => s.trim()).filter(Boolean),
   408→      media: effective.media.filter((m) => m.url.trim()),
   409→      order: Number.isFinite(effective.order) ? Math.max(0, Math.round(effective.order)) : 99,
   410→    };
   411→    setSaving(true);
   412→    try {
   413→      if (isNew) {
   414→        const res = await api<{ project: Project }>("api/projects", { body, admin: true });
   415→        toast.success("Project saved");
   416→        navigate(`/admin/projects/${res.project.id}`);
   417→      } else {
   418→        await api<{ project: Project }>(`api/projects/${id}`, { method: "PATCH", body, admin: true });
   419→        setForm((f) => ({ ...f, published: body.published }));
   420→        toast.success("Project saved");
   421→        navigate("/admin/projects");
   422→      }
   423→    } catch (err) {
   424→      onApiError(err);
   425→    } finally {
   426→      setSaving(false);
   427→    }
   428→  }
   429→
   430→  async function remove() {
   431→    if (!id) return;
   432→    setDeleting(true);
   433→    try {
   434→      await api<{ success: boolean }>(`api/projects/${id}`, { method: "DELETE", admin: true });
   435→      toast.success("Project deleted");
   436→      navigate("/admin/projects");
   437→    } catch (err) {
   438→      onApiError(err);
   439→    } finally {
   440→      setDeleting(false);
   441→    }
   442→  }
   443→
   444→  if (loading) {
   445→    return (
   446→      <div className="mx-auto max-w-3xl space-y-4">
   447→        <div className="h-8 w-48 animate-pulse rounded bg-ink/10" />
   448→        <div className="h-64 animate-pulse rounded-xl bg-ink/10" />
   449→        <div className="h-64 animate-pulse rounded-xl bg-ink/10" />
   450→      </div>
   451→    );
   452→  }
   453→
   454→  if (loadError) {
   455→    return (
   456→      <div className="mx-auto max-w-3xl space-y-4">
   457→        <ErrorNote message={loadError} />
   458→        <Button variant="outline" onClick={() => navigate("/admin/projects")}>
   459→          Back to projects
   460→        </Button>
   461→      </div>
   462→    );
   463→  }
   464→
   465→  return (
   466→    <div className="mx-auto max-w-3xl pb-28">
   467→      <PageHead
   468→        title={isNew ? "New project" : "Edit project"}
   469→        desc={isNew ? "Fill in the story — you can save as a draft and publish later." : `/${form.slug}`}
   470→      >
   471→        <Button variant="outline" className="h-10" onClick={() => navigate("/admin/projects")}>
   472→          Back to list
   473→        </Button>
   474→      </PageHead>
   475→
   476→      <div className="space-y-6">
   477→        {/* basics */}
   478→        <AdminCard>
   479→          <CardHead title="Basics" desc="Title, slug and classification." />
   480→          <div className="space-y-4 p-5">
   481→            <Field label="Title" htmlFor="p-title">
   482→              <Input
   483→                id="p-title"
   484→                value={form.title}
   485→                placeholder="e.g. Brew & Co. Website"
   486→                onChange={(e) => {
   487→                  const title = e.target.value;
   488→                  setForm((f) => ({
   489→                    ...f,
   490→                    title,
   491→                    slug: slugTouched ? f.slug : slugify(title),
   492→                  }));
   493→                }}
   494→              />
   495→            </Field>
   496→            <Field
   497→              label="Slug"
   498→              htmlFor="p-slug"
   499→              hint={slugError ?? "Public URL: #/work/<slug> — lowercase letters, numbers, dashes."}
   500→            >
   501→              <Input
   502→                id="p-slug"
   503→                value={form.slug}
   504→                className={cn(slugError && "border-destructive")}
   505→                onChange={(e) => {
   506→                  setSlugTouched(true);
   507→                  set("slug", e.target.value);
   508→                }}
   509→              />
   510→            </Field>
   511→            <div className="grid gap-4 sm:grid-cols-2">
   512→              <Field label="Client (optional)" htmlFor="p-client">
   513→                <Input id="p-client" value={form.client} placeholder="e.g. Brew & Co." onChange={(e) => set("client", e.target.value)} />
   514→              </Field>
   515→              <Field label="Year" htmlFor="p-year">
   516→                <Input id="p-year" value={form.year} placeholder="2025" onChange={(e) => set("year", e.target.value)} />
   517→              </Field>
   518→              <Field label="Category" htmlFor="p-category">
   519→                <Select value={form.category} onValueChange={(v) => set("category", v)}>
   520→                  <SelectTrigger id="p-category" className="h-9">
   521→                    <SelectValue />
   522→                  </SelectTrigger>
   523→                  <SelectContent>
   524→                    {CATEGORIES.map((c) => (
   525→                      <SelectItem key={c} value={c}>
   526→                        {c}
   527→                      </SelectItem>
   528→                    ))}
   529→                  </SelectContent>
   530→                </Select>
   531→              </Field>
   532→              <Field
   533→                label="Project type"
   534→                htmlFor="p-type"
   535→                hint="Label sample work as Concept Project until it's genuine client work."
   536→              >
   537→                <Select value={form.projectType} onValueChange={(v) => set("projectType", v)}>
   538→                  <SelectTrigger id="p-type" className="h-9">
   539→                    <SelectValue />
   540→                  </SelectTrigger>
   541→                  <SelectContent>
   542→                    {PROJECT_TYPES.map((t) => (
   543→                      <SelectItem key={t} value={t}>
   544→                        {t}
   545→                      </SelectItem>
   546→                    ))}
   547→                  </SelectContent>
   548→                </Select>
   549→              </Field>
   550→            </div>
   551→            <Field label="Live URL (optional)" htmlFor="p-liveurl" hint="Link to the deployed site, if it's public.">
   552→              <Input id="p-liveurl" value={form.liveUrl} placeholder="https://…" onChange={(e) => set("liveUrl", e.target.value)} />
   553→            </Field>
   554→          </div>
   555→        </AdminCard>
   556→
   557→        {/* story */}
   558→        <AdminCard>
   559→          <CardHead title="Story" desc="The narrative shown on the public case study." />
   560→          <div className="space-y-4 p-5">
   561→            <Field label="Short description" htmlFor="p-desc" hint="One or two lines used in cards and previews.">
   562→              <CountedTextarea id="p-desc" value={form.description} onChange={(v) => set("description", v)} max={400} rows={3} />
   563→            </Field>
   564→            <Field label="Overview" htmlFor="p-overview">
