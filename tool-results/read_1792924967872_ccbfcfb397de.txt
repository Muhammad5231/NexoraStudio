    32→export function SettingsView() {
    33→  const [settings, setSettings] = useState<Settings | null>(null);
    34→  const [loading, setLoading] = useState(true);
    35→  const [error, setError] = useState<string | null>(null);
    36→
    37→  const { onApiError } = useAdmin();
    38→  const loadStable = useCallback(async () => {
    39→    setLoading(true);
    40→    setError(null);
    41→    try {
    42→      const res = await api<{ settings: Settings }>("api/settings", { admin: true });
    43→      setSettings(res.settings);
    44→    } catch (err) {
    45→      if (onApiError(err)) return;
    46→      setError(err instanceof Error ? err.message : "Could not load settings.");
    47→    } finally {
    48→      setLoading(false);
    49→    }
    50→  }, [onApiError]);
    51→
    52→  useEffect(() => {
    53→    void loadStable();
    54→  }, [loadStable]);
    55→
    56→  if (loading) {
    57→    return (
    58→      <div>
    59→        <PageHead title="Settings" desc="Everything editable on the public site." />
    60→        <div className="space-y-4">
    61→          <Skeleton className="h-10 w-full max-w-xl" />
    62→          <Skeleton className="h-64 w-full" />
    63→        </div>
    64→      </div>
    65→    );
    66→  }
    67→
    68→  if (error || !settings) {
    69→    return (
    70→      <div className="space-y-4">
    71→        <ErrorNote message={error ?? "Settings unavailable."} />
    72→        <Button variant="outline" onClick={() => void loadStable()}>
    73→          Try again
    74→        </Button>
    75→      </div>
    76→    );
    77→  }
    78→
    79→  return (
    80→    <div>
    81→      <PageHead title="Settings" desc="Everything editable on the public site — saved per section." />
    82→      <Tabs defaultValue="brand">
    83→        <TabsList className="mb-5 h-auto w-full flex-wrap justify-start gap-1 bg-ink/5 p-1">
    84→          <TabsTrigger value="brand">Brand</TabsTrigger>
    85→          <TabsTrigger value="contact">Contact</TabsTrigger>
    86→          <TabsTrigger value="social">Social</TabsTrigger>
    87→          <TabsTrigger value="seo">SEO</TabsTrigger>
    88→          <TabsTrigger value="home">Home</TabsTrigger>
    89→          <TabsTrigger value="about">About</TabsTrigger>
    90→          <TabsTrigger value="footer">Footer</TabsTrigger>
    91→          <TabsTrigger value="pages">Pages</TabsTrigger>
    92→          <TabsTrigger value="account">Account</TabsTrigger>

   127→/* ------------------------------------------------------------------ shared */
   128→/** Form state that re-seeds whenever the server value identity changes.
   129→ *  Uses the documented "adjust state during render" pattern (no effect). */
   130→function useSyncedForm<T>(value: T) {
   131→  const [form, setForm] = useState<T>(value);
   132→  const [seed, setSeed] = useState<T>(value);
   133→  if (seed !== value) {
   134→    setSeed(value);
   135→    setForm(value);
   136→  }
   137→  return [form, setForm] as const;
   138→}
   139→
   140→function useSectionSaver<K extends keyof Settings>(section: K, onSaved: (s: Settings) => void) {
   141→  const { onApiError } = useAdmin();
   142→  const [saving, setSaving] = useState(false);
   143→
   144→  async function save(data: Settings[K]) {
   145→    setSaving(true);
   146→    try {
   147→      const res = await api<{ settings: Settings }>("api/settings", {
   148→        method: "PUT",
   149→        body: { section, data },
   150→        admin: true,
   151→      });
   152→      onSaved(res.settings);
   153→      toast.success("Settings saved");
   154→    } catch (err) {
   155→      onApiError(err);
   156→    } finally {
   157→      setSaving(false);
   158→    }
   159→  }
   160→
   161→  return { saving, save };
   162→}

   173→function SaveBar({
   174→  onSave,
   175→  saving,
   176→  dirty,
   177→}: {
   178→  onSave: () => void;
   179→  saving: boolean;
   180→  dirty: boolean;
   181→}) {
   182→  return (
   183→    <div className="flex items-center justify-end gap-3">
   184→      {!dirty ? <span className="text-xs text-graphite">No changes yet</span> : <span className="text-xs font-medium text-ink">Unsaved changes</span>}

   186→        {saving ? <Spin /> : null} Save changes
   187→      </Button>
   188→    </div>
   189→  );
   190→}

   546→/* ----------------------------------------------------------------- account */
   547→function AccountSection() {
   548→  const [current, setCurrent] = useState("");
   549→  const [next, setNext] = useState("");
   550→  const [confirm, setConfirm] = useState("");
   551→  const [saving, setSaving] = useState(false);
   552→  const [error, setError] = useState<string | null>(null);
   553→
   554→  async function submit(e: FormEvent) {
   555→    e.preventDefault();
   556→    setError(null);
   557→    if (!current) return setError("Enter your current password.");
   558→    if (!PASSWORD_RE.test(next))
   559→      return setError("New password needs at least 8 characters, including a letter and a number.");
   560→    if (next !== confirm) return setError("New password and confirmation don't match.");
   561→    setSaving(true);
   562→    try {
   563→      await api("api/auth", { body: { action: "password", currentPassword: current, newPassword: next } });
   564→      toast.success("Password updated");
   565→      setCurrent("");
   566→      setNext("");
   567→      setConfirm("");
   568→    } catch (err) {
   569→      setError(err instanceof Error ? err.message : "Could not update password.");
   570→    } finally {
   571→      setSaving(false);
   572→    }
   573→  }
   574→
   575→  return (
   576→    <div className="max-w-xl space-y-4">
   577→      <AdminCard>
   578→        <CardHead title="Change password" desc="Use at least 8 characters with a letter and a number." />
   579→        <form onSubmit={(e) => void submit(e)} className="space-y-4 p-5" noValidate>
   580→          <Field label="Current password" htmlFor="acc-current">
   581→            <Input id="acc-current" type="password" autoComplete="current-password" value={current} onChange={(e) => setCurrent(e.target.value)} />
   582→          </Field>
   583→          <Field label="New password" htmlFor="acc-next">
   584→            <Input id="acc-next" type="password" autoComplete="new-password" value={next} onChange={(e) => setNext(e.target.value)} />
   585→          </Field>
   586→          <Field label="Confirm new password" htmlFor="acc-confirm">
   587→            <Input id="acc-confirm" type="password" autoComplete="new-password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
   588→          </Field>
   589→          {error ? (
   590→            <p role="alert" className="rounded-lg border border-destructive/30 bg-destructive/5 px-3 py-2 text-sm text-destructive">
   591→              {error}
   592→            </p>
   593→          ) : null}
   594→          <Button type="submit" className="h-10 bg-ink text-ivory hover:bg-ink/85" disabled={saving}>
   595→            {saving ? <Spin /> : null} Update password
   596→          </Button>
   597→        </form>
   598→      </AdminCard>
   599→    </div>
   600→  );
   601→}
